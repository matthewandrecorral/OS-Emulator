cmake_minimum_required(VERSION 3.16)
project(OS_Emulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(emulator_core Scheduler.cpp Marquee.cpp CommandInterpreter.cpp)
target_include_directories(emulator_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(emulator_core PUBLIC Threads::Threads)
if(WIN32)
 set(CONSOLE_SOURCE ConsoleWindows.cpp)
elseif(APPLE OR UNIX)
 set(CONSOLE_SOURCE ConsolePosix.cpp)
else()
 message(FATAL_ERROR "Unsupported terminal platform")
endif()
add_executable(OS-Emulator main.cpp ConsoleUI.cpp ${CONSOLE_SOURCE})
target_link_libraries(OS-Emulator PRIVATE emulator_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(emulator_tests tests/emulator_tests.cpp ConsoleUI.cpp)
 target_link_libraries(emulator_tests PRIVATE emulator_core)
 add_test(NAME scheduler_and_commands COMMAND emulator_tests)
 set_tests_properties(scheduler_and_commands PROPERTIES TIMEOUT 30)
endif()
